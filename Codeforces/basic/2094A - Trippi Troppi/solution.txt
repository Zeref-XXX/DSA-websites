#include <bits/stdc++.h>
using namespace std;
int main()
{
    int n;
    cin >> n;
    cin.ignore();
    while (n--)
    {
        string s;
        getline(cin, s);
        int a = s.length();
        string ans = string(1, s[0]);
        for (int i = 1; i < a; i++)
        {
            if (s[i - 1] == ' ')
            {
                ans.push_back(s[i]);
            }
        }
        cout << ans << endl;
    }
    return 0;
}
