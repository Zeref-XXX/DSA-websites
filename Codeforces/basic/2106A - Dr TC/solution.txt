#include <bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin >> t;
    while (t--)
    {
        int n;cin >> n;
        string s;cin >> s;
        int count=0;
        for(int i=0;i<n;i++){
            string temp=s;
            temp[i] = (temp[i] == '1') ? '0' : '1';
            
            for(int j=0;j<n;j++){
                if(temp[j]=='1') count++;
            }
            
        }
        cout<<count<<endl;
    }
}
