#include <bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin >> t;
    while (t--)
    {
        int n, m, x, y;
        cin >> n >> m >> x >> y;
        vector<int> n_c;
        for (int i = 0; i < n; i++)
        {
            int temp;
            cin >> temp;
            n_c.push_back(temp);
        }
        vector<int>m_c;
        for(int i=0;i<m;i++){
            int temp;cin>>temp;
            m_c.push_back(temp);
        }
        cout<<m_c.size()+n_c.size()<<endl;
    }
}
