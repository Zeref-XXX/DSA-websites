#include <bits/stdc++.h>
using namespace std;

int main()
{

    int t;
    cin >> t;

    while (t--)
    {
        int index = 1;
        int n;
        cin >> n; 
        vector<int> v(3 * n);

        for (int i = 0; i < 3 * n; i = i + 3)
        {
            v[i] = index++;
        }
        for (int i = 0; i < 3 * n; i++)
        {
            if (i % 3 != 0)
                v[i] = index++;
        }
       
        for (int i = 0; i < 3 * n; i++)
        {
            cout << v[i]<<" ";
        }
        cout << endl;
    }
}
