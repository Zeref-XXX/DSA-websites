#include <bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin >> t;
    while (t--)
    {
        vector<int> vec;
        for (int i = 0; i < 7; i++)
        {
            int temp;
            cin >> temp;
            vec.push_back(temp);
        }
        sort(vec.begin(), vec.end());
        int ans = 0;
        for (int i = 0; i < 6; i++)
        {
            ans += vec[i];
        }
        ans = -1 * ans+vec[6];
        cout<<ans<<endl;
    }
}
